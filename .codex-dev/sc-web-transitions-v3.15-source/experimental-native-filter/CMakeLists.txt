cmake_minimum_required(VERSION 3.28)
project(sc-transitions-frame-probe VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(libobs CONFIG REQUIRED)
add_library(sc-transitions-frame-probe MODULE sc-native-probe.c)
target_link_libraries(sc-transitions-frame-probe PRIVATE OBS::libobs)
set_target_properties(sc-transitions-frame-probe PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(sc-transitions-frame-probe PRIVATE /W4)
else()
  target_compile_options(sc-transitions-frame-probe PRIVATE -Wall -Wextra)
endif()
